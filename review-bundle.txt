const mongoose = require('mongoose');

const bookingSchema = new mongoose.Schema({
  slotId:            { type: mongoose.Schema.Types.ObjectId, ref: 'Slot',    required: true },
  customerId:        { type: mongoose.Schema.Types.ObjectId, ref: 'User',    required: true },
  vendorId:          { type: mongoose.Schema.Types.ObjectId, ref: 'User',    required: true },
  serviceId:         { type: mongoose.Schema.Types.ObjectId, ref: 'Service', required: true },
  status:            { type: String, enum: ['pending', 'confirmed', 'cancelled', 'failed'], default: 'pending' },
  paymentStatus:     { type: String, enum: ['pending', 'paid', 'failed'], default: 'pending' },
  amount:            { type: Number },
  currency:          { type: String, default: 'INR' },
  razorpayOrderId:   { type: String },
  razorpayPaymentId: { type: String },
  razorpaySignature: { type: String },
  expiresAt:         { type: Date }
}, { timestamps: true });

/**
 * DOUBLE-BOOKING PREVENTION — Partial Unique Index
 *
 * Covers bookings with status in ['pending', 'confirmed'].
 * Prevents concurrent holds and double-bookings on the same slot.
 * Cancelled or failed bookings drop out of the partial index filter,
 * immediately freeing the slot for other customers.
 */
bookingSchema.index(
  { slotId: 1 },
  { unique: true, partialFilterExpression: { status: { $in: ['pending', 'confirmed'] } } }
);

bookingSchema.index({ razorpayOrderId: 1 });

module.exports = mongoose.model('Booking', bookingSchema);

// Vendor creates available slots for one of their services.
// Ownership check: service must belong to the requesting vendor.
const createSlot = async (req, res, next) => {
  try {
    const { serviceId, startTime, endTime } = req.body;

    const service = await Service.findById(serviceId);
    if (!service) return res.status(404).json({ message: 'Service not found' });
    if (String(service.vendorId) !== String(req.user._id)) {
      return res.status(403).json({ message: 'Not authorized to create slots for this service' });
    }

    // Validate time ordering
    if (new Date(endTime) <= new Date(startTime)) {
      return res.status(400).json({ message: 'endTime must be after startTime' });
    }

    const slot = await Slot.create({
      vendorId: req.user._id,
      serviceId,
      startTime,
      endTime
    });
    res.status(201).json(slot);
  } catch (err) { next(err); }
};

// ─── POST /api/bookings ───────────────────────────────────────────────────────
// Customer books a slot (creates pending booking with 10-minute hold window).
// Double-booking guard: partial unique index on (slotId) WHERE status IN ('pending', 'confirmed').
// Only when payment is verified (via /verify or webhook) does status flip to 'confirmed'
// and paymentStatus to 'paid'.
const bookSlot = async (req, res, next) => {
  try {
    await releaseExpiredHolds();
    const { slotId } = req.body;

    // Verify slot exists
    const slot = await Slot.findById(slotId);
    if (!slot) return res.status(404).json({ message: 'Slot not found' });

    // Reject if slot is currently blocked by walk-in or offline pause
    if (slot.isBlocked) {
      return res.status(409).json({ message: 'This slot is currently blocked by the vendor (walk-in customer or temporary pause)' });
    }

    // Soft pre-check: reject if slot is already reserved or booked
    if (slot.status !== 'available') {
      return res.status(409).json({ message: 'Slot is already booked or reserved' });
    }

    const service = await Service.findById(slot.serviceId);
    if (!service) return res.status(404).json({ message: 'Service not found for this slot' });

    const amount = (service.price !== undefined ? service.price : 500) * 100; // in paise
    const expiresAt = new Date(Date.now() + 10 * 60 * 1000); // 10-min hold

    // Attempt the Booking insert — partial unique index rejects concurrent holds (11000)
    let booking;
    try {
      booking = await Booking.create({
        slotId,
        customerId:    req.user._id,
        vendorId:      slot.vendorId,
        serviceId:     slot.serviceId,
        amount,
        currency:      'INR',
        status:        'pending',
        paymentStatus: 'pending',
        expiresAt
      });
    } catch (dbError) {
      if (dbError.code === 11000) {
        return res.status(409).json({ message: 'Slot is already booked or reserved (concurrent request)' });
      }
      throw dbError;
    }

    // Atomically flip slot status to 'reserved' ensuring it was not blocked or reserved in parallel
    const reservedSlot = await Slot.findOneAndUpdate(
      { _id: slotId, status: 'available', isBlocked: { $ne: true } },
      { status: 'reserved' }
    );
    if (!reservedSlot) {
      await booking.deleteOne();
      return res.status(409).json({ message: 'Slot was reserved or blocked by the vendor during checkout initiation' });
    }

    const populated = await booking.populate('slotId serviceId vendorId', '-password');
    res.status(201).json(populated);
  } catch (err) { next(err); }
};

// ─── PATCH /api/bookings/:id/cancel ──────────────────────────────────────────
// Cancel a booking (customer who booked OR vendor who owns the service).
// Freeing the slot: sets slot.status back to 'available' AND cancels the
// booking. Because the partial index only covers status IN ['pending', 'confirmed'],
// the slot is immediately re-bookable after cancellation.
const cancelBooking = async (req, res, next) => {
  try {
    const booking = await Booking.findById(req.params.id);
    if (!booking) return res.status(404).json({ message: 'Booking not found' });

    const isCustomer = String(booking.customerId) === String(req.user._id);
    const isVendor   = String(booking.vendorId)   === String(req.user._id);

    if (!isCustomer && !isVendor) {
      return res.status(403).json({ message: 'Not authorized to cancel this booking' });
    }
    if (booking.status === 'cancelled') {
      return res.status(400).json({ message: 'Booking is already cancelled' });
    }

    // 1. Cancel the booking and mark paymentStatus failed if unpaid
    booking.status = 'cancelled';
    if (booking.paymentStatus === 'pending') {
      booking.paymentStatus = 'failed';
    }
    await booking.save();

    // 2. Free the slot — atomically
    await Slot.findByIdAndUpdate(booking.slotId, { status: 'available' });

    const populated = await Booking.findById(booking._id)
      .populate('slotId', 'startTime endTime status')
      .populate('serviceId', 'name price')
      .populate('vendorId', 'name email')
      .populate('customerId', 'name email');

    res.json({ message: 'Booking cancelled successfully', booking: populated });
  } catch (err) { next(err); }
};

const crypto = require('crypto');
const Booking = require('../models/Booking');
const Slot = require('../models/Slot');
const Service = require('../models/Service');

const { releaseExpiredHolds } = require('../utils/bookingCleanup');

// ─── POST /api/payments/create-order ─────────────────────────────────────────
// Customer creates a Razorpay order for their pending booking.
// Ownership check: must be customer who owns the booking.
const createOrder = async (req, res, next) => {
  try {
    await releaseExpiredHolds();
    const { bookingId } = req.body;
    if (!bookingId) {
      return res.status(400).json({ message: 'bookingId is required' });
    }

    const booking = await Booking.findById(bookingId).populate('serviceId slotId');
    if (!booking) {
      return res.status(404).json({ message: 'Booking not found' });
    }

    // Ownership check
    if (String(booking.customerId) !== String(req.user._id)) {
      return res.status(403).json({ message: 'Not authorized for this booking' });
    }

    // Check if already paid
    if (booking.paymentStatus === 'paid') {
      return res.status(400).json({ message: 'Booking is already paid and confirmed' });
    }

    // Check if cancelled or failed
    if (booking.status === 'cancelled' || booking.status === 'failed') {
      return res.status(400).json({ message: `Cannot pay for a ${booking.status} booking` });
    }

    // Check expiration of temporary hold (10 min)
    if (booking.expiresAt && new Date() > new Date(booking.expiresAt)) {
      booking.status = 'failed';
      booking.paymentStatus = 'failed';
      await booking.save();
      await Slot.findByIdAndUpdate(booking.slotId._id, { status: 'available' });
      return res.status(410).json({ message: 'Booking hold expired. Please select the slot again.' });
    }

    // Calculate price in paise (1 INR = 100 paise)
    const servicePrice = booking.serviceId?.price !== undefined ? booking.serviceId.price : 500;
    const amountInPaise = servicePrice * 100;

    const paymentMode = process.env.PAYMENT_MODE || 'razorpay';
    const orderOptions = {
      amount: amountInPaise,
      currency: 'INR',
      receipt: `b_${String(booking._id).slice(-10)}`,
      notes: {
        bookingId: String(booking._id),
        customerId: String(req.user._id)
      }
    };

    let order;
    if (paymentMode === 'simulated') {
      order = {
        id: `order_sim_${String(booking._id).slice(-12)}`,
        amount: amountInPaise,
        currency: 'INR',
        isSimulated: true
      };
    } else {
      try {
        const razorpay = getRazorpayInstance();
        order = await razorpay.orders.create(orderOptions);
      } catch (rzpErr) {
        console.error(`Razorpay orders.create failed: ${rzpErr.message || 'API error'}`);
        return res.status(503).json({ message: 'Payment service temporarily unavailable. Please try again later.' });
      }
    }

    booking.razorpayOrderId = order.id;
    booking.amount = amountInPaise;
    await booking.save();

    res.status(200).json({
      orderId: order.id,
      amount: order.amount,
      currency: order.currency,
      keyId: process.env.RAZORPAY_KEY_ID || 'rzp_test_simulated',
      bookingId: booking._id,
      isSimulated: !!order.isSimulated
    });
  } catch (err) {
    next(err);
  }
};

// ─── POST /api/payments/verify ───────────────────────────────────────────────
// Receives Razorpay's payment response from frontend.
// Verifies HMAC SHA256 signature server-side.
// ONLY marks booking as paid and confirmed if signature is valid.
const verifyPayment = async (req, res, next) => {
  try {
    const { bookingId, razorpay_order_id, razorpay_payment_id, razorpay_signature } = req.body;

    if (!bookingId || !razorpay_order_id || !razorpay_payment_id || !razorpay_signature) {
      return res.status(400).json({
        message: 'Missing required fields: bookingId, razorpay_order_id, razorpay_payment_id, and razorpay_signature are required'
      });
    }

    const booking = await Booking.findById(bookingId);
    if (!booking) {
      return res.status(404).json({ message: 'Booking not found' });
    }

    // Ownership check
    if (String(booking.customerId) !== String(req.user._id)) {
      return res.status(403).json({ message: 'Not authorized for this booking' });
    }

    // Idempotency: if already verified and paid
    if (booking.paymentStatus === 'paid' && booking.status === 'confirmed') {
      const populated = await Booking.findById(booking._id)
        .populate('slotId', 'startTime endTime status')
        .populate('serviceId', 'name price')
        .populate('vendorId', 'name email');
      return res.status(200).json({
        success: true,
        message: 'Payment already verified and booking confirmed',
        booking: populated
      });
    }

    // Verify order ID match if set
    if (booking.razorpayOrderId && booking.razorpayOrderId !== razorpay_order_id) {
      return res.status(400).json({ message: 'Order ID does not match this booking' });
    }

    // Cryptographic signature verification using server-side Razorpay secret
    const secret = process.env.RAZORPAY_KEY_SECRET || 'simulated_test_secret';

    const expectedSignature = crypto
      .createHmac('sha256', secret)
      .update(`${razorpay_order_id}|${razorpay_payment_id}`)
      .digest('hex');

    const expectedBuf = Buffer.from(expectedSignature, 'utf-8');
    const receivedBuf = Buffer.from(razorpay_signature, 'utf-8');

    const paymentMode = process.env.PAYMENT_MODE || 'razorpay';
    const isSimulatedValid = paymentMode === 'simulated' && (
      razorpay_order_id.startsWith('order_sim_') &&
      (razorpay_signature === 'simulated_valid_signature' || razorpay_signature === expectedSignature)
    );

    const isValid = isSimulatedValid || (
      expectedBuf.length === receivedBuf.length &&
      crypto.timingSafeEqual(expectedBuf, receivedBuf)
    );

    if (!isValid) {
      return res.status(400).json({
        success: false,
        message: 'Invalid payment signature. Verification failed.'
      });
    }

    // Signature verified: update booking and slot status
    booking.paymentStatus = 'paid';
    booking.status = 'confirmed';
    booking.razorpayPaymentId = razorpay_payment_id;
    booking.razorpaySignature = razorpay_signature;
    await booking.save();

    await Slot.findByIdAndUpdate(booking.slotId, { status: 'booked' });

    const populated = await Booking.findById(booking._id)
      .populate('slotId', 'startTime endTime status')
      .populate('serviceId', 'name price')
      .populate('vendorId', 'name email');

    res.status(200).json({
      success: true,
      message: 'Payment verified successfully and booking confirmed',
      booking: populated
    });
  } catch (err) {
    next(err);
  }
};

// ─── POST /api/payments/webhook ──────────────────────────────────────────────
// Razorpay webhook endpoint — independent source of truth for payment confirmation.
// Verifies webhook signature against raw request body using RAZORPAY_WEBHOOK_SECRET.
const handleWebhook = async (req, res, next) => {
  try {
    const webhookSignature = req.headers['x-razorpay-signature'];
    const webhookSecret = process.env.RAZORPAY_WEBHOOK_SECRET;

    if (!webhookSecret) {
      console.error('RAZORPAY_WEBHOOK_SECRET is not configured');
      return res.status(500).json({ message: 'Webhook secret not configured on server' });
    }

    if (!webhookSignature) {
      return res.status(400).json({ message: 'Missing x-razorpay-signature header' });
    }

    // Use rawBody Buffer captured by express.json({ verify }) or fallback
    const rawPayload = req.rawBody || Buffer.from(JSON.stringify(req.body), 'utf-8');

    const expectedSignature = crypto
      .createHmac('sha256', webhookSecret)
      .update(rawPayload)
      .digest('hex');

    const expectedBuf = Buffer.from(expectedSignature, 'utf-8');
    const receivedBuf = Buffer.from(webhookSignature, 'utf-8');

    const isValid = (
      expectedBuf.length === receivedBuf.length &&
      crypto.timingSafeEqual(expectedBuf, receivedBuf)
    );

    if (!isValid) {
      return res.status(400).json({ message: 'Invalid webhook signature' });
    }

    const event = req.body.event;
    const payload = req.body.payload;

    if (event === 'payment.captured' || event === 'order.paid') {

      }

        booking.status = 'confirmed';

        await Slot.findByIdAndUpdate(booking.slotId, { status: 'booked' });

    } else if (event === 'payment.failed') {

      let booking = null;

        await Slot.findByIdAndUpdate(booking.slotId, { status: 'available' });
        console.log(`[Webhook] Booking ${booking._id} marked failed via ${event}; slot freed`);

/**
 * Release pending holds that have expired past their 10-minute window,
 * and automatically unblock slots whose walk-in pause window has passed.

const releaseExpiredHolds = async () => {
  try {
    const now = new Date();

    // 1. Release expired checkout holds
    const expiredBookings = await Booking.find({
      status: 'pending',
      paymentStatus: 'pending',
      expiresAt: { $lt: now }
    });

    for (const booking of expiredBookings) {
      booking.status = 'failed';
      booking.paymentStatus = 'failed';
      await booking.save();
      await Slot.findByIdAndUpdate(booking.slotId, { status: 'available' });
    }

    // 2. Automatically unblock slots whose temporary walk-in pause has expired
    await Slot.updateMany(
      {
        isBlocked: true,
        blockedUntil: { $lte: now }
      },
      {
        $set: { isBlocked: false },
        $unset: { blockedUntil: 1, blockedReason: 1 }
      }
    );

    return expiredBookings.length;
  } catch (err) {
    console.error('Error in releaseExpiredHolds:', err);
    return 0;
  }
};

  cyan:   (s) => `\x1b[36m${s}\x1b[0m`,
  bold:   (s) => `\x1b[1m${s}\x1b[0m`,
  dim:    (s) => `\x1b[2m${s}\x1b[0m`,
};

let passed = 0;
let failed = 0;

function assert(condition, label) {
  if (condition) {
    console.log(`  ${c.green('✓')} ${label}`);
    passed++;
  } else {
    console.log(`  ${c.red('✗')} ${label}`);
    failed++;
  }
}

async function runTests() {
  console.log(c.bold(c.cyan('\n══════════════════════════════════════════════════════════════')));
  console.log(c.bold(c.cyan('  Bug 1 — Simulated Signature Bypass — Fix Verification')));
  console.log(c.bold(c.cyan('══════════════════════════════════════════════════════════════\n')));

  let mongoServer, server;

  try {
    mongoServer = await MongoMemoryServer.create();
    await mongoose.connect(mongoServer.getUri());
    console.log(c.dim('  Connected to in-memory MongoDB.\n'));

    const app = express();
    app.use(cors());
    app.use(express.json());
    app.use('/api', bookingRoutes);
    app.use('/api/payments', paymentRoutes);
    app.use(errorHandler);

    server = http.createServer(app);
    await new Promise((resolve) => server.listen(0, resolve));
    const port = server.address().port;
    const baseUrl = `http://127.0.0.1:${port}/api`;

    async function api(endpoint, { method = 'GET', token, body } = {}) {
      const h = { 'Content-Type': 'application/json' };
      if (token) h['Authorization'] = `Bearer ${token}`;
      const res = await fetch(`${baseUrl}${endpoint}`, {
        method, headers: h,
        body: body ? JSON.stringify(body) : undefined
      });
      return { status: res.status, data: await res.json() };
    }

    // Helper: create a fresh slot + book it + return { bookingId, slotId }
    let slotCounter = 0;
    async function bookFreshSlot(customerToken, vendor, service) {
      slotCounter++;
      const slot = await Slot.create({
        vendorId: vendor._id, serviceId: service._id,
        startTime: new Date(`2026-12-${String(slotCounter).padStart(2, '0')}T10:00:00Z`),
        endTime:   new Date(`2026-12-${String(slotCounter).padStart(2, '0')}T11:00:00Z`),
        status: 'available'
      });
      const bookRes = await api('/bookings', {
        method: 'POST', token: customerToken,
        body: { slotId: slot._id.toString() }
      });
      return { bookingId: bookRes.data._id, slotId: slot._id };
    }

    // Seed
    const vendor = await User.create({ name: 'Vendor', email: 'v@b1.com', password: 'x', role: 'vendor' });
    const customer = await User.create({ name: 'Customer', email: 'c@b1.com', password: 'x', role: 'customer' });
    const token = jwt.sign({ id: customer._id }, process.env.JWT_SECRET, { expiresIn: '1h' });
    const service = await Service.create({ name: 'Test Service', price: 500, vendorId: vendor._id });

    // ══════════════════════════════════════════════════════════════════════════
    // Test A: razorpay mode + Razorpay API failure → 503
    // ══════════════════════════════════════════════════════════════════════════
    console.log(c.bold('Test A — razorpay mode + Razorpay API failure → 503'));
    process.env.PAYMENT_MODE = 'razorpay';
    mockShouldFail = true;

    const { bookingId: bookingA } = await bookFreshSlot(token, vendor, service);
    const orderResA = await api('/payments/create-order', {
      method: 'POST', token, body: { bookingId: bookingA }
    });

    assert(orderResA.status === 503, `createOrder returns 503 (got ${orderResA.status})`);
    assert(orderResA.data.message.includes('temporarily unavailable'), 'Error message says service unavailable');
    mockShouldFail = false;

    // ══════════════════════════════════════════════════════════════════════════
    // Test B: razorpay mode + magic string on order_sim_ ID → REJECTED
    // ══════════════════════════════════════════════════════════════════════════
    console.log('\n' + c.bold('Test B — razorpay mode + magic string on order_sim_ ID → 400'));
    process.env.PAYMENT_MODE = 'razorpay';

    const { bookingId: bookingB } = await bookFreshSlot(token, vendor, service);
    // Manually inject an order_sim_ ID (as if it leaked from dev or an earlier fallback)
    await Booking.findByIdAndUpdate(bookingB, { razorpayOrderId: 'order_sim_LEAKED_ID' });

    const verifyB = await api('/payments/verify', {
      method: 'POST', token,
      body: {
        bookingId: bookingB,
        razorpay_order_id: 'order_sim_LEAKED_ID',
        razorpay_payment_id: 'pay_FAKE',
        razorpay_signature: 'simulated_valid_signature'   // ← THE EXPLOIT
      }
    });

    assert(verifyB.status === 400, `Magic string REJECTED with 400 (got ${verifyB.status})`);
    const bB = await Booking.findById(bookingB);
    assert(bB.paymentStatus === 'pending', `Booking still pending (got ${bB.paymentStatus})`);

    // ══════════════════════════════════════════════════════════════════════════
    // Test C: razorpay mode + valid HMAC → ACCEPTED
    // ══════════════════════════════════════════════════════════════════════════
    console.log('\n' + c.bold('Test C — razorpay mode + valid HMAC → 200'));
    process.env.PAYMENT_MODE = 'razorpay';

    const { bookingId: bookingC, slotId: slotC } = await bookFreshSlot(token, vendor, service);
    const orderResC = await api('/payments/create-order', {
      method: 'POST', token, body: { bookingId: bookingC }
    });
    const orderIdC = orderResC.data.orderId;
    const paymentIdC = 'pay_real_valid_123';

    const validSig = crypto
      .createHmac('sha256', process.env.RAZORPAY_KEY_SECRET)
      .update(`${orderIdC}|${paymentIdC}`)
      .digest('hex');

    const verifyC = await api('/payments/verify', {
      method: 'POST', token,
      body: {
        bookingId: bookingC,
        razorpay_order_id: orderIdC,
        razorpay_payment_id: paymentIdC,
        razorpay_signature: validSig
      }
    });

    assert(verifyC.status === 200, `Valid HMAC accepted with 200 (got ${verifyC.status})`);
    const bC = await Booking.findById(bookingC);
    assert(bC.paymentStatus === 'paid', 'Booking marked paid');
    assert(bC.status === 'confirmed', 'Booking confirmed');
    const sC = await Slot.findById(slotC);
    assert(sC.status === 'booked', 'Slot marked booked');

    // ══════════════════════════════════════════════════════════════════════════
    // Test D: razorpay mode + invalid HMAC → REJECTED
    // ══════════════════════════════════════════════════════════════════════════
    console.log('\n' + c.bold('Test D — razorpay mode + invalid HMAC → 400'));
    process.env.PAYMENT_MODE = 'razorpay';

    const { bookingId: bookingD } = await bookFreshSlot(token, vendor, service);
    const orderResD = await api('/payments/create-order', {
      method: 'POST', token, body: { bookingId: bookingD }
    });

    const verifyD = await api('/payments/verify', {
      method: 'POST', token,
      body: {
        bookingId: bookingD,
        razorpay_order_id: orderResD.data.orderId,
        razorpay_payment_id: 'pay_real_456',
        razorpay_signature: 'totally_wrong_signature_deadbeef'
      }
    });

    assert(verifyD.status === 400, `Invalid HMAC rejected with 400 (got ${verifyD.status})`);
    const bD = await Booking.findById(bookingD);
    assert(bD.paymentStatus === 'pending', `Booking still pending (got ${bD.paymentStatus})`);

    // ══════════════════════════════════════════════════════════════════════════
    // Test E: simulated mode + magic string → ACCEPTED
    // ══════════════════════════════════════════════════════════════════════════
    console.log('\n' + c.bold('Test E — simulated mode + magic string → 200'));
    process.env.PAYMENT_MODE = 'simulated';

    const { bookingId: bookingE, slotId: slotE } = await bookFreshSlot(token, vendor, service);
    const orderResE = await api('/payments/create-order', {
      method: 'POST', token, body: { bookingId: bookingE }
    });

    assert(orderResE.data.orderId.startsWith('order_sim_'), `Got simulated order: ${orderResE.data.orderId}`);

    const verifyE = await api('/payments/verify', {
      method: 'POST', token,
      body: {
        bookingId: bookingE,
        razorpay_order_id: orderResE.data.orderId,
        razorpay_payment_id: 'pay_sim_789',
        razorpay_signature: 'simulated_valid_signature'
      }
    });

    assert(verifyE.status === 200, `Simulated mode accepts magic string with 200 (got ${verifyE.status})`);
    const bE = await Booking.findById(bookingE);
    assert(bE.paymentStatus === 'paid', 'Booking marked paid in simulated mode');
    assert(bE.status === 'confirmed', 'Booking confirmed in simulated mode');
    const sE = await Slot.findById(slotE);
    assert(sE.status === 'booked', 'Slot marked booked in simulated mode');

    // Reset for other tests
    process.env.PAYMENT_MODE = 'razorpay';

    // ── Summary ─────────────────────────────────────────────────────────────
    console.log('\n' + c.bold(c.cyan('══════════════════════════════════════════════════════════════')));
    if (failed === 0) {
      console.log(c.bold(c.green(`  ✅ ALL ${passed} ASSERTIONS PASSED`)));
      console.log(c.green('  razorpay mode: API failure → 503          VERIFIED'));
      console.log(c.green('  razorpay mode: magic string → REJECTED    VERIFIED'));
      console.log(c.green('  razorpay mode: valid HMAC → ACCEPTED      VERIFIED'));
      console.log(c.green('  razorpay mode: invalid HMAC → REJECTED    VERIFIED'));
      console.log(c.green('  simulated mode: magic string → ACCEPTED   VERIFIED'));
    } else {
      console.log(c.bold(c.red(`  ❌ ${failed} ASSERTION(S) FAILED (${passed} passed)`)));
    }
    console.log(c.bold(c.cyan('══════════════════════════════════════════════════════════════\n')));

    process.exitCode = failed === 0 ? 0 : 1;

  } catch (err) {
    console.error(c.red('\nFatal test error:'), err);
    process.exitCode = 1;
  } finally {
    if (server) server.close();
    await mongoose.disconnect();
    if (mongoServer) await mongoServer.stop();
  }
}

runTests();
